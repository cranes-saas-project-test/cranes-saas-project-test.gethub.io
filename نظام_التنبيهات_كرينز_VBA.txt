-- جدول التنبيهات
CREATE TABLE tbl_Alerts (
    AlertID AUTOINCREMENT PRIMARY KEY,
    AlertType TEXT(50),
    CraneID LONG,
    FaultID LONG,
    RelatedForm TEXT(50),
    AlertMessage TEXT(255),
    AlertDate DATETIME,
    IsRead YESNO DEFAULT False
);

-- استعلام عرض التنبيهات
CREATE VIEW qry_Alerts AS
SELECT * FROM tbl_Alerts ORDER BY AlertDate DESC;

'==============================================================================
' وحدة التنبيهات العامة - Module: mod_Alerts
'==============================================================================
Option Compare Database
Option Explicit

Public Sub CreateAlert( _
    ByVal AlertType As String, _
    ByVal CraneID As Variant, _
    ByVal FaultID As Variant, _
    ByVal RelatedForm As String, _
    ByVal AlertMessage As String)

    Dim sql As String
    Dim db As DAO.Database

    Set db = CurrentDb

    ' بناء استعلام الإضافة
    sql = "INSERT INTO tbl_Alerts " & _
          "(AlertType, CraneID, FaultID, RelatedForm, AlertMessage, AlertDate, IsRead) VALUES (" & _
          "'" & AlertType & "', " & _
          IIf(IsNull(CraneID), "Null", CraneID) & ", " & _
          IIf(IsNull(FaultID), "Null", FaultID) & ", " & _
          "'" & RelatedForm & "', " & _
          "'" & Replace(AlertMessage, "'", "''") & "', Now(), False)"

    db.Execute sql

    Set db = Nothing

    ' إشعار فوري
    If DCount("*", "tbl_Alerts", "IsRead=False") > 0 Then
        ' يمكن إضافة صوت تنبيه هنا
    End If
End Sub

'==============================================================================
' دالة تحديث حالة التنبيهات
'==============================================================================

Public Sub MarkAlertAsRead(ByVal AlertID As Long)
    Dim sql As String
    sql = "UPDATE tbl_Alerts SET IsRead = True WHERE AlertID = " & AlertID
    CurrentDb.Execute sql
End Sub

'==============================================================================
' دالة جلب عدد التنبيهات غير المقروءة
'==============================================================================

Public Function GetUnreadAlertsCount() As Long
    GetUnreadAlertsCount = DCount("*", "tbl_Alerts", "IsRead=False")
End Function

'==============================================================================
' دالة جلب آخر تنبيه
'==============================================================================

Public Function GetLastAlertMessage() As String
    Dim rs As DAO.Recordset
    Set rs = CurrentDb.OpenRecordset("SELECT TOP 1 AlertMessage FROM tbl_Alerts WHERE IsRead=False ORDER BY AlertDate DESC")

    If Not rs.EOF Then
        GetLastAlertMessage = rs!AlertMessage
    Else
        GetLastAlertMessage = ""
    End If

    rs.Close
    Set rs = Nothing
End Function

Private Sub txtSearch_Change()
    If Len(Me.txtSearch.Text) > 0 Then
        Me.Filter = "AlertMessage LIKE '*" & Me.txtSearch.Text & "*'"
        Me.FilterOn = True
    Else
        Me.FilterOn = False
    End If
End Sub

'==============================================================================
' فلترة حسب الحالة
'==============================================================================

Private Sub cmbFilter_AfterUpdate()
    Select Case Me.cmbFilter.Value
        Case 1 ' الكل
            Me.FilterOn = False
        Case 2 ' غير مقروء
            Me.Filter = "IsRead=False"
            Me.FilterOn = True
        Case 3 ' مقروء
            Me.Filter = "IsRead=True"
            Me.FilterOn = True
    End Select
End Sub

'==============================================================================
' زر حفظ العطل وإنشاء تنبيه
'******************************************************************************

Private Sub btnSaveFault_Click()
    ' التحقق من البيانات
    If IsNull(Me.cmbCraneID) Then
        MsgBox "الرجاء اختيار الرافعة", vbExclamation
        Me.cmbCraneID.SetFocus
        Exit Sub
    End If

    If IsNull(Me.txtFaultDescription) Then
        MsgBox "الرجاء إدخال وصف العطل", vbExclamation
        Me.txtFaultDescription.SetFocus
        Exit Sub
    End If

    ' حفظ السجل
    If Me.Dirty Then
        Me.Dirty = False
    End If

    ' إنشاء تنبيه
    Call CreateAlert( _
        "عطل", _
        Me.cmbCraneID, _
        Me.FaultID, _
        "frm_CraneFaults", _
        "🚨 عطل جديد في الرافعة رقم " & Me.cmbCraneID & vbCrLf & _
        "الوصف: " & Me.txtFaultDescription)

    MsgBox "✅ تم حفظ العطل وإرسال التنبيه", vbInformation

    ' تحديث الداشبورد
    If CurrentProject.AllForms("frm_Dashboard").IsLoaded Then
        DoCmd.Close acForm, "frm_Dashboard", acSaveNo
        DoCmd.OpenForm "frm_Dashboard"
    End If
End Sub

'==============================================================================
' منع الصرف عند عدم كفاية الرصيد + تنبيه
'******************************************************************************

Private Sub Form_BeforeUpdate(Cancel As Integer)
    Dim AvailableQty As Double
    Dim RequestedQty As Double

    RequestedQty = Me.txtQuantity

    ' حساب الرصيد المتاح
    AvailableQty = GetAvailableBalance(Me.cmbItemID, Me.cmbStoreID)

    If RequestedQty > AvailableQty Then
        MsgBox _
        "❌ لا يمكن الصرف!" & vbCrLf & _
        "الكمية المطلوبة: " & RequestedQty & vbCrLf & _
        "الرصيد المتاح: " & AvailableQty, vbCritical, "خطأ في الصرف"

        Cancel = True
        Exit Sub
    End If
End Sub

'==============================================================================
' بعد حفظ الصرف - إضافة تلقائية لمخزن المخلفات
'****************************************************************************==

Private Sub Form_AfterInsert()
    Const STORE_SPAREPARTS As Long = 1  ' مخزن قطع الغيار
    Const STORE_SCRAP As Long = 5        ' مخزن المخلفات

    ' إذا كان الصرف من مخزن قطع الغيار
    If Me.cmbStoreID = STORE_SPAREPARTS Then
        ' إضافة تلقائية للمخلفات
        Call CreateAlert( _
            "تحويل مخلفات", _
            Null, _
            Null, _
            "frm_StoreOut", _
            "♻️ تم تحويل " & Me.txtQuantity & " منقطع غيار إلى مخزن المخلفات")
    End If

    ' إذا كانت الكمية قرب حد النقص
    Dim AvailableQty As Double
    AvailableQty = GetAvailableBalance(Me.cmbItemID, Me.cmbStoreID)

    If AvailableQty <= GetReorderLevel(Me.cmbItemID) Then
        Call CreateAlert( _
            "نقص مخزون", _
            Null, _
            Null, _
            "frm_Items", _
            "⚠️ الصنف [" & Me.cmbItemID & "] وصل لحد النقص! الرصيد: " & AvailableQty)
    End If
End Sub

'==============================================================================
' دوال مساعدة للصرف
'==============================================================================

Private Function GetAvailableBalance(ItemID As Long, StoreID As Long) As Double
    Dim OpeningBal As Double
    Dim TotalIn As Double
    Dim TotalOut As Double

    OpeningBal = Nz(DSum("OpeningBalance", "tbl_ItemOpeningBalance", _
        "ItemID=" & ItemID & " AND StoreID=" & StoreID), 0)

    TotalIn = Nz(DSum("Quantity", "tbl_StoreIn", _
        "ItemID=" & ItemID & " AND StoreID=" & StoreID), 0)

    TotalOut = Nz(DSum("Quantity", "tbl_StoreOut", _
        "ItemID=" & ItemID & " AND StoreID=" & StoreID), 0)

    GetAvailableBalance = OpeningBal + TotalIn - TotalOut
End Function

Private Function GetReorderLevel(ItemID As Long) As Double
    GetReorderLevel = Nz(DLookup("ReorderLevel", "tbl_Items", "ItemID=" & ItemID), 0)
End Function

'==============================================================================
' تسجيل الدخول
'******************************************************************************

Private Sub btnLogin_Click()
    Dim rs As DAO.Recordset
    Dim strUser As String
    Dim strPass As String

    strUser = Nz(Me.txtUserName, "")
    strPass = Nz(Me.txtPassword, "")

    If strUser = "" Or strPass = "" Then
        MsgBox "الرجاء إدخال اسم المستخدم وكلمة المرور", vbExclamation
        Exit Sub
    End If

    ' البحث عن المستخدم
    Set rs = CurrentDb.OpenRecordset( _
        "SELECT * FROM tbl_Users WHERE UserName='" & strUser & "' AND Password='" & strPass & "'")

    If rs.EOF Then
        MsgBox "اسم المستخدم أو كلمة المرور غير صحيحة", vbExclamation
        Me.txtPassword = ""
        Me.txtPassword.SetFocus
        rs.Close
        Set rs = Nothing
        Exit Sub
    End If

    ' التحقق من حالة الحساب
    If rs!IsActive = False Then
        MsgBox "هذا الحساب موقوف", vbCritical
        rs.Close
        Set rs = Nothing
        Exit Sub
    End If

    ' حفظ بيانات المستخدم الحالي
    g_CurrentUserID = rs!UserID
    g_CurrentUserName = rs!UserName
    g_CurrentUserRole = rs!Role

    rs.Close
    Set rs = Nothing

    ' فتح الداشبورد
    DoCmd.Close acForm, Me.Name
    DoCmd.OpenForm "frm_Dashboard"
End Sub

Public Function HasPermission(PermissionName As String) As Boolean
    Dim rs As DAO.Recordset

    Set rs = CurrentDb.OpenRecordset( _
        "SELECT * FROM tbl_UserPermissions UP " & _
        "INNER JOIN tbl_Permissions P ON UP.PermissionID = P.PermissionID " & _
        "WHERE UP.UserID=" & g_CurrentUserID & " AND P.PermissionName='" & PermissionName & "'")

    HasPermission = Not rs.EOF

    rs.Close
    Set rs = Nothing
End Function
